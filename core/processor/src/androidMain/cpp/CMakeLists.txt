cmake_minimum_required(VERSION 3.22.1)
project(formsnap_processor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED OPENCV_ANDROID_SDK OR OPENCV_ANDROID_SDK STREQUAL "")
    message(FATAL_ERROR "OPENCV_ANDROID_SDK is required.")
endif()
set(OpenCV_DIR "${OPENCV_ANDROID_SDK}/sdk/native/jni")
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_library(formsnap_processor SHARED native-lib.cpp)
target_include_directories(formsnap_processor PRIVATE "${OpenCV_INCLUDE_DIRS}")
target_link_libraries(formsnap_processor PRIVATE ${OpenCV_LIBS} android log)
set_target_properties(formsnap_processor PROPERTIES POSITION_INDEPENDENT_CODE ON)
